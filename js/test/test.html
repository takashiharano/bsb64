#----------------------------------------------------------
test set id "bsb64"

${i} = 0
LOOP (${i} < $TEST_DATA.length) (
  ${ret} = bsb64.encode($TEST_DATA[${i}].src, $TEST_DATA[${i}].n)
  test verify ${ret} == $TEST_DATA[${i}].b64
  ${i}++
)

${i} = 0
LOOP (${i} < $TEST_DATA.length) (
  ${ret} = bsb64.decodeToString($TEST_DATA[${i}].b64, $TEST_DATA[${i}].n)
  test verify ${ret} == $TEST_DATA[${i}].src
  ${i}++
)

test end
